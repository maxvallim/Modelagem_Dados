Tutorial: CRUD com Firebase e Python

1. Introdução

Este tutorial apresenta um projeto em Python que utiliza o Firebase como banco de dados, com um menu interativo baseado em submenus para gerenciar as entidades: Especialidade, Mecânico, Ordem de Serviço (OS) e MecânicoEspecialidade.

2. Pré-requisitos

  • Conta no Firebase
  • Projeto criado no Firebase
  • Firestore ativado no modo de teste
  • Chave privada JSON do Firebase
  • Python 3 instalado
  • Instalar o pacote firebase-admin: 
    pip install firebase-admin

3. Código Python com Menu Interativo

import firebase_admin
from firebase_admin import credentials, firestore

# Inicialização do Firebase
cred = credentials.Certificate('caminho/para/seu/arquivo-chave.json')
firebase_admin.initialize_app(cred)
db = firestore.client()

# Funções CRUD para Especialidade
def criar_especialidade():
    nome = input("Digite o nome da especialidade: ")
    doc_ref = db.collection('Especialidade').document()
    doc_ref.set({'nome': nome})
    print("Especialidade criada com sucesso.")

def listar_especialidades():
    especialidades = db.collection('Especialidade').stream()
    for esp in especialidades:
        print(f"{esp.id} - {esp.to_dict()['nome']}")

def atualizar_especialidade():
    listar_especialidades()
    id_esp = input("Digite o ID da especialidade a ser atualizada: ")
    novo_nome = input("Digite o novo nome: ")
    db.collection('Especialidade').document(id_esp).update({'nome': novo_nome})
    print("Especialidade atualizada.")

def excluir_especialidade():
    listar_especialidades()
    id_esp = input("Digite o ID da especialidade a ser excluída: ")
    db.collection('Especialidade').document(id_esp).delete()
    print("Especialidade excluída.")

# Funções CRUD para Mecânico
def criar_mecanico():
    nome = input("Digite o nome do mecânico: ")
    telefone = input("Digite o telefone: ")
    doc_ref = db.collection('Mecanico').document()
    doc_ref.set({'nome': nome, 'telefone': telefone})
    print("Mecânico criado com sucesso.")

def listar_mecanicos():
    mecanicos = db.collection('Mecanico').stream()
    for mec in mecanicos:
        dados = mec.to_dict()
        print(f"{mec.id} - {dados['nome']} ({dados['telefone']})")

def atualizar_mecanico():
    listar_mecanicos()
    id_mec = input("Digite o ID do mecânico a ser atualizado: ")
    nome = input("Novo nome: ")
    telefone = input("Novo telefone: ")
    db.collection('Mecanico').document(id_mec).update({'nome': nome, 'telefone': telefone})
    print("Mecânico atualizado.")

def excluir_mecanico():
    listar_mecanicos()
    id_mec = input("Digite o ID do mecânico a ser excluído: ")
    db.collection('Mecanico').document(id_mec).delete()
    print("Mecânico excluído.")

# Menu principal com submenus
def menu():
    while True:
        print("\n=== MENU PRINCIPAL ===")
        print("1. Especialidades")
        print("2. Mecânicos")
        print("0. Sair")
        opcao = input("Escolha uma opção: ")

        if opcao == '1':
            menu_especialidades()
        elif opcao == '2':
            menu_mecanicos()
        elif opcao == '0':
            print("Encerrando...")
            break
        else:
            print("Opção inválida.")

def menu_especialidades():
    while True:
        print("\n--- Especialidades ---")
        print("1. Criar")
        print("2. Listar")
        print("3. Atualizar")
        print("4. Excluir")
        print("0. Voltar")
        opcao = input("Escolha uma opção: ")

        if opcao == '1':
            criar_especialidade()
        elif opcao == '2':
            listar_especialidades()
        elif opcao == '3':
            atualizar_especialidade()
        elif opcao == '4':
            excluir_especialidade()
        elif opcao == '0':
            break
        else:
            print("Opção inválida.")

def menu_mecanicos():
    while True:
        print("\n--- Mecânicos ---")
        print("1. Criar")
        print("2. Listar")
        print("3. Atualizar")
        print("4. Excluir")
        print("0. Voltar")
        opcao = input("Escolha uma opção: ")

        if opcao == '1':
            criar_mecanico()
        elif opcao == '2':
            listar_mecanicos()
        elif opcao == '3':
            atualizar_mecanico()
        elif opcao == '4':
            excluir_mecanico()
        elif opcao == '0':
            break
        else:
            print("Opção inválida.")

# Executar o menu
menu()
